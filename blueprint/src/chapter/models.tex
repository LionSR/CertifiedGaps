\chapter{Further models}
\label{chap:models}

The reflection-positive ring criterion applies across spin-1 and spin-half chains and ladders, both
for Hamiltonians that are reflection positive in a natural real basis (the computational basis for
the Ising models, the Cartesian basis for spin 1) and for antiferromagnetic models that become
reflection positive after a sublattice rotation. Each model in
\texttt{CertifiedGaps/Models/*.lean} is defined as a reflection-positive ring together with its
certified parameter points, seed inverse temperature \(\beta_0\), and rational chain certificate,
giving a uniform gap bound \(\log r / \beta_0\) at every point by Theorem~\ref{thm:rp-gap}. The
only open hypothesis for each model is a single lemma stating its two seed inequalities at every
certified point, which are verified numerically from complete eigenvalue enclosures of the seed
rings in the corresponding directory under \texttt{models/}. Ring lengths count unit cells.

\section{Local building blocks}

The spin-\(\tfrac12\) and dimerized models share three local operators: the real spin-\(\tfrac12\)
bond factors, the bipartite sublattice rotation \(e^{i\pi S^y}\), which on one site of a bond turns
the
antiferromagnetic exchange into minus a sum of squares of real factors, and the two-site cell swap
\(F\) used as the internal reflection on
dimerized rings.

\begin{definition}[Real Pauli and spin-\(\tfrac12\) factors]
  \label{def:local-spin-half}
  \lean{CertifiedGaps.pauliZ, CertifiedGaps.pauliX, CertifiedGaps.pauliZ_transpose,
    CertifiedGaps.pauliX_transpose, CertifiedGaps.SpinHalf.P, CertifiedGaps.SpinHalf.P_symm}
  \leanok
  Let \(Z = \operatorname{diag}(1, -1)\) and \(X\) be the real symmetric Pauli matrices, and let
  \(P_\alpha = (S^x, iS^y, S^z)\) be the real spin-\(\tfrac12\) factors, each of which is
  symmetric or antisymmetric.
\end{definition}

\begin{lemma}[Sublattice rotation]
  \label{lem:spin-half-rotation}
  \lean{CertifiedGaps.SpinHalf.spinHalf, CertifiedGaps.SpinHalf.rot,
    CertifiedGaps.SpinHalf.rot_mul_conjTranspose, CertifiedGaps.SpinHalf.rot_conj_spinHalf,
    CertifiedGaps.SpinHalf.Pc_eq}
  \leanok
  \uses{def:local-spin-half}
  The rotation \(R = e^{i\pi S^y} = \bigl(\begin{smallmatrix} 0 & 1 \\ -1 & 0
  \end{smallmatrix}\bigr)\)
  is unitary and satisfies \(R S^\alpha R^\dagger = s_\alpha S^\alpha\) with \(s = (-1, 1, -1)\).
  Since the real factors satisfy \(P_\alpha = c_\alpha S^\alpha\) with \(c = (1, i, 1)\), rotating
  one site of the exchange interaction \(\mathbf S \cdot \mathbf S\) turns it into
  \(-\sum_\alpha P_\alpha \otimes P_\alpha\).
\end{lemma}
\begin{proof}
  \leanok
  Evaluate both sides entrywise on \(2 \times 2\) matrices.
\end{proof}

\begin{definition}[Cell swap]
  \label{def:cell-swap}
  \lean{CertifiedGaps.cellSwap, CertifiedGaps.cellSwap_mul_kronecker_mul_cellSwap,
    CertifiedGaps.cellSwap_mul_cellSwap, CertifiedGaps.cellSwap_transpose,
    CertifiedGaps.cellSwap_conj_smul_sum_kronecker_self}
  \leanok
  The swap \(F\) of the two sites of a cell is a real symmetric involution satisfying
  \(F (A \otimes B) F = B \otimes A\) and commuting with every intra-cell exchange
  \(c \sum_k A_k \otimes A_k\). It serves as the internal cell reflection of the dimerized models.
\end{definition}

\section{Spin-1 chain with single-ion anisotropy}

\section{Ising chain in transverse and longitudinal fields}

\section{Spin-\(\tfrac12\) Heisenberg chain in a staggered field}

odd site turns each exchange \(\mathbf S \cdot \mathbf S\) into
\(-S^xS^x + S^yS^y - S^zS^z = -\sum_\alpha P_\alpha \otimes P_\alpha\) and the staggered field into
a
uniform field, producing a reflection-positive ring with the same spectrum.

reflection-positive ring of two-site rungs.

  The rotated Heisenberg ladder is a ring of two-site rungs with bond factors
  \(P_\alpha \otimes 1\) and \(1 \otimes P_\alpha\) and rung exchange as the on-site term
  \(o = -J_\perp \sum_\alpha P_\alpha \otimes P_\alpha\). It is certified at rung coupling
  \(J_\perp = 4\) with seed length \(n_0 = 4\) rungs, \(\beta_0 = 2\), and \(r = 9.033\)
  (\texttt{models/heisenberg\_ladder}).
\end{definition}

\begin{lemma}
  \label{lem:ladder-valid}
  \lean{CertifiedGaps.HeisenbergLadder.points_valid, CertifiedGaps.HeisenbergLadder.points_β₀_pos}
  \leanok
  \uses{def:ladder}
  The certificate is valid and \(\beta_0 > 0\).
\end{lemma}
\begin{proof}
  \leanok
  This is checked by \texttt{decide +kernel}.
\end{proof}

\begin{lemma}[Seeds]
  \label{lem:ladder-seeds}
  \lean{CertifiedGaps.HeisenbergLadder.seeds}
  \leanok
  \uses{def:ladder, def:rp-purities}
  The seed purities satisfy \(\Ps(4, 2) \ge 1 - 0.03509\) and \(\Pt(8, 2) \ge 1 - 0.02463\).
\end{lemma}
\begin{proof}
  These bounds are certified in \texttt{models/heisenberg\_ladder} and are left as an open
  numerical hypothesis in Lean.
\end{proof}

\begin{theorem}
  \label{thm:ladder-gap}
  \lean{CertifiedGaps.HeisenbergLadder.gap}
  \leanok
  \uses{lem:ladder-valid, lem:ladder-seeds, thm:rp-gap}
  Every even ring of \(L \ge 4\) rungs of the rotated ladder has a unique ground state and
  \(\Delta_L > \log 9.033 / 2 \approx 1.100\).
\end{theorem}
\begin{proof}
  \leanok
  Apply Theorem~\ref{thm:rp-gap}.
\end{proof}

\begin{definition}[Physical ladder]
  \label{def:ladder-physical}
  \lean{CertifiedGaps.HeisenbergLadder.legS, CertifiedGaps.HeisenbergLadder.physicalHamiltonian,
    CertifiedGaps.HeisenbergLadder.physicalSpectrum, CertifiedGaps.HeisenbergLadder.rungRot}
  \leanok
  \uses{def:ring-operators, lem:spin-half-rotation}
  The physical ladder has Hamiltonian
  \(H = \sum_{\text{legs}} \mathbf S_i \cdot \mathbf S_j + J_\perp \sum_{\text{rungs}}
  \mathbf S_i \cdot \mathbf S_j\) and is related to the rotated ladder by the checkerboard rotation
  \(e^{i\pi S^y}\) applied to site \((1, j)\) for odd \(j\) and site \((2, j)\) for even \(j\).
\end{definition}

\begin{lemma}
  \label{lem:ladder-physical-eq}
  \lean{CertifiedGaps.HeisenbergLadder.physicalHamiltonian_eq,
    CertifiedGaps.HeisenbergLadder.physicalSpectrum_eq}
  \leanok
  \uses{def:ladder-physical, def:ladder, thm:conj-hamiltonian, thm:conj-spectrum}
  On a ring with an even number of rungs, the physical ladder is unitarily conjugate to the rotated
  ladder, so their spectra agree.
\end{lemma}
\begin{proof}
  \leanok
  On an even ring, the checkerboard rotation acts on one endpoint of every leg bond and one
  endpoint of every rung bond, mapping each exchange interaction to
  \(-\sum_\alpha P_\alpha \otimes P_\alpha\) by Lemma~\ref{lem:spin-half-rotation}.
\end{proof}

\begin{theorem}
  \label{thm:ladder-physical-gap}
  \lean{CertifiedGaps.HeisenbergLadder.physical_gap}
  \leanok
  \uses{lem:ladder-physical-eq, thm:ladder-gap}
  Every even ring of \(L \ge 4\) rungs of the physical ladder at \(J_\perp = 4\) has a unique
  ground state and \(\Delta_L > 1.100\).
\end{theorem}
\begin{proof}
  \leanok
  Combine Lemma~\ref{lem:ladder-physical-eq} with Theorem~\ref{thm:ladder-gap}.
\end{proof}
