\chapter{The spin-1 Heisenberg ring}
\label{chap:heisenberg}

Haldane's conjecture~\cite{Haldane1983Continuum, Haldane1983Nonlinear} states that integer-spin
antiferromagnetic Heisenberg chains have a unique ground state and a nonzero spectral gap in the
thermodynamic limit (originally established rigorously for the valence-bond-solid parent
Hamiltonian in~\cite{Affleck1987Rigorous}). Applying the spacetime-bootstrap criterion of
Chapters~\ref{chap:criterion}--\ref{chap:rings} to the periodic spin-1 antiferromagnetic Heisenberg
ring~\eqref{eq:heisenberg-ham} (Definition~\ref{def:heisenberg}) shows that for every even
\(L \ge 2304\) the ground state is unique and the spectral gap satisfies
\(\Delta_L > (\log 20)/784 \approx 0.003821\) (Theorem~\ref{thm:haldane-gap}), and that for every
even \(L \ge 60\) it satisfies \(\Delta_L > \tfrac{4}{105}\log\tfrac{80}{79}\)
(Theorem~\ref{thm:haldane-gap-sixty}). These are the two bounds of Theorem~1.1
of~\cite{OpenAIHaldane}. In this chapter each bound assumes one table of short-ring traces:
\texttt{TraceTable} at \(\beta = 49/4\) or \texttt{TraceTable60} at \(\beta = 21/2\).
The trial states are checked in the kernel (Lemma~\ref{lem:kernel-spin1-trials}).
Each table follows from the hypothesis that an integer recurrence reproduces the totals printed
in~\cite[Appendices A and B]{OpenAIHaldane}: eighteen integers (\texttt{HornerTotals}) and nine
integers (\texttt{HornerTotals60}). The kernel cannot recompute most of these integers at
the paper's sizes; it checks the two periodic totals at length \(4\)
(Lemma~\ref{lem:kernel-horner-four}). So both gaps
hold given only the printed integers (Theorems~\ref{thm:kernel-gap36-integers}
and~\ref{thm:kernel-gap60-integers}), which also give sharper constants. The older theorems of
\texttt{Main.lean}, whose seeds are \texttt{sorry}, are not used.

\section{The spin-1 Heisenberg Hamiltonian}

In the Cartesian basis, the spin-1 operators are purely imaginary antisymmetric matrices defined
directly from the Levi-Civita symbol (Definition~\ref{def:heisenberg}), so the main theorem
statements (Theorems~\ref{thm:haldane-gap} and~\ref{thm:haldane-gap-sixty}) refer only to standard
spin-1 exchange matrices without introducing the reflection-positive ring formalism of
Definition~\ref{def:rpring}.

\begin{definition}[Spin-1 Heisenberg ring Hamiltonian]
  \label{def:heisenberg}
  \lean{CertifiedGaps.SpinOne.leviCivita, CertifiedGaps.SpinOne.S, CertifiedGaps.SpinOne.exchange,
    CertifiedGaps.SpinOneHeisenberg.hamiltonian}
  \leanok
  \uses{def:ring-operators}
  In the Cartesian basis, the spin-1 matrices have entries \((S^\alpha)_{ss'} = -i\,\epsilon_{\alpha
    s s'}\)
  given by the Levi-Civita symbol \(\epsilon\). The two-site exchange operator is
  \(\mathbf S \cdot \mathbf S = \sum_\alpha S^\alpha \otimes S^\alpha\), and the ring Hamiltonian on
  \((\C^3)^{\otimes L}\) with periodic boundary conditions (Definition~\ref{def:ring-operators}) is
  \begin{equation}
    \label{eq:heisenberg-ham}
    H_L = \sum_{j=0}^{L-1} \mathbf S_j \cdot \mathbf S_{j+1}.
  \end{equation}
\end{definition}

\begin{definition}[Eigenvalue spectrum, partition function, and purities]
  \label{def:heisenberg-spectrum}
  \lean{CertifiedGaps.SpinOneHeisenberg.hamiltonian_isHermitian,
    CertifiedGaps.SpinOneHeisenberg.spectrum, CertifiedGaps.SpinOneHeisenberg.partitionFunction,
    CertifiedGaps.SpinOneHeisenberg.spatialPurity, CertifiedGaps.SpinOneHeisenberg.thermalPurity}
  \leanok
  \uses{def:heisenberg}
  The Hamiltonian \(H_L\) in~\eqref{eq:heisenberg-ham} is Hermitian with eigenvalue family \(E_L\),
  partition function \(Z_L(\beta) = \operatorname{Re}\Tr e^{-\beta H_L}\), and purities \(\Ps\) and
  \(\Pt\) formed from these traces as in Definition~\ref{def:purities}.
\end{definition}

\begin{lemma}[Boltzmann expansion of the partition function]
  \label{lem:heisenberg-trace}
  \lean{CertifiedGaps.SpinOneHeisenberg.trace_exp_hamiltonian,
    CertifiedGaps.SpinOneHeisenberg.partitionFunction_eq_sum,
    CertifiedGaps.SpinOneHeisenberg.partitionFunction_spectrum}
  \leanok
  \uses{def:heisenberg-spectrum, lem:ring-trace-exp}
  The matrix trace in Definition~\ref{def:heisenberg-spectrum} satisfies
  \(Z_L(\beta) = \sum_m e^{-\beta E_L(m)}\), matching the Boltzmann partition function of
  Definition~\ref{def:partition-function}.
\end{lemma}
\begin{proof}
  \leanok
  Apply Lemma~\ref{lem:ring-trace-exp} to the Hermitian matrix \(H_L\) of
  Definition~\ref{def:heisenberg-spectrum}.
\end{proof}

\section{Reflection positivity in the Cartesian spin-1 basis}

Because \(G_\alpha = iS^\alpha\) is real and antisymmetric in the Cartesian basis
(Definition~\ref{def:spin-one-generators}), the negated exchange interaction~\eqref{eq:spin-one-rp}
is a sum of tensor squares of real matrices, so the Heisenberg chain and its single-ion anisotropic
extension (Definition~\ref{def:spin-one-large-d} and Lemma~\ref{lem:large-d-heisenberg}) are
reflection-positive rings in the sense of Definition~\ref{def:rpring}
(Theorem~\ref{thm:heisenberg-moments}).

\begin{definition}[Cartesian spin-1 generators]
  \label{def:spin-one-generators}
  \lean{CertifiedGaps.SpinOne.G, CertifiedGaps.SpinOne.G_transpose}
  \leanok
  \uses{def:heisenberg}
  Each generator \(G_\alpha = i S^\alpha\) has real antisymmetric entries \((G_\alpha)_{ss'} =
  \epsilon_{\alpha s s'}\), and
  \begin{equation}
    \label{eq:spin-one-rp}
    -\mathbf S \cdot \mathbf S = \sum_\alpha G_\alpha \otimes G_\alpha.
  \end{equation}
\end{definition}

\begin{definition}[Spin-1 ring with single-ion anisotropy]
  \label{def:spin-one-large-d}
  \lean{CertifiedGaps.SpinOneLargeD.Sz2, CertifiedGaps.SpinOneLargeD.model}
  \leanok
  \uses{def:rpring, def:spin-one-generators}
  For single-ion anisotropy \(D \in \R\), the anisotropic spin-1 ring is the reflection-positive
  ring (Definition~\ref{def:rpring}) with bond operators \(C_\alpha = G_\alpha\) from
  Definition~\ref{def:spin-one-generators}, on-site term
  \(o = D\,(S^z)^2 = D \operatorname{diag}(1, 1, 0)\), and internal cell reflection \(F = 1\).
\end{definition}

\begin{lemma}[Isotropic limit of the single-ion anisotropic ring]
  \label{lem:large-d-heisenberg}
  \lean{CertifiedGaps.SpinOneLargeD.bondMatrix_eq, CertifiedGaps.SpinOneLargeD.oc_eq,
    CertifiedGaps.SpinOneLargeD.hamiltonian_eq, CertifiedGaps.SpinOneLargeD.partitionFunction_zero}
  \leanok
  \uses{def:spin-one-large-d, def:heisenberg-spectrum}
  The bond operator~\eqref{eq:rp-bond} of the ring in Definition~\ref{def:spin-one-large-d} equals
  \(\mathbf S \cdot \mathbf S\) and its Hamiltonian~\eqref{eq:rp-hamiltonian} is
  \(\sum_j \mathbf S_j \cdot \mathbf S_{j+1} + D \sum_j (S^z_j)^2\). At \(D = 0\), its partition
  functions coincide with those of the isotropic spin-1 Heisenberg ring
  (Definition~\ref{def:heisenberg-spectrum}).
\end{lemma}
\begin{proof}
  \leanok
  The bond identity~\eqref{eq:spin-one-rp}, namely
  \(-\sum_\alpha G_\alpha \otimes G_\alpha = -\sum_\alpha (iS^\alpha) \otimes (iS^\alpha) =
  \sum_\alpha S^\alpha \otimes S^\alpha\), holds entrywise from \(i^2 = -1\). At \(D = 0\) the
  on-site anisotropy terms vanish, reducing~\eqref{eq:rp-hamiltonian} to~\eqref{eq:heisenberg-ham}.
\end{proof}

\begin{theorem}[Transfer-operator moment inequalities for the Heisenberg ring]
  \label{thm:heisenberg-moments}
  \lean{CertifiedGaps.SpinOneHeisenberg.spatialMomentBounds}
  \leanok
  \uses{lem:large-d-heisenberg, thm:rp-moment-bounds}
  For every \(\beta > 0\), the partition function \(L \mapsto Z_L(\beta)\) of the spin-1 Heisenberg
  ring (Definition~\ref{def:heisenberg-spectrum}) satisfies the transfer-operator moment
  inequalities (1)--(3) of Definition~\ref{def:moment-bounds} above length \(2\).
\end{theorem}
\begin{proof}
  \leanok
  Apply Theorem~\ref{thm:rp-moment-bounds} to the anisotropic ring of
  Definition~\ref{def:spin-one-large-d} at \(D = 0\), whose partition functions agree with those of
  the Heisenberg ring by Lemma~\ref{lem:large-d-heisenberg}.
\end{proof}

\section{Rotations, integer matrices, and short-ring spectra}

The Cartesian basis proves reflection positivity, while the \(S^z\) basis makes the finite
recurrence integral. The two descriptions are related by exact unitary identities.

\begin{lemma}[Rotating a spin-1 bond]
  \label{lem:spin1-rotations}
  \lean{CertifiedGaps.SpinOne.PreservesCross, CertifiedGaps.SpinOne.rotExchange,
    CertifiedGaps.SpinOne.conj_S_of_preservesCross, CertifiedGaps.SpinOne.conj_rotExchange}
  \leanok
  \uses{def:heisenberg}
  Define \(h(R)=\sum_{\alpha,\gamma}R_{\alpha\gamma}S^\alpha\otimes S^\gamma\).
  If a real matrix \(Q\) preserves the cross product, then
  \(Q^*S^\alpha Q=\sum_\gamma Q_{\alpha\gamma}S^\gamma\) and
  \((Q\otimes Q)^*h(R)(Q\otimes Q)=h(Q^{\mathsf T}RQ)\).
\end{lemma}
\begin{proof}
  \leanok
  Expand the Levi-Civita coefficients in the cross-product identity.
\end{proof}

\begin{lemma}[The integer \(S^z\) frame]
  \label{lem:spin1-sz}
  \lean{CertifiedGaps.SpinOneHeisenberg.exchangeZ,
    CertifiedGaps.SpinOneHeisenberg.szFrame_conjTranspose_mul_self,
    CertifiedGaps.SpinOneHeisenberg.conj_hamiltonian,
    CertifiedGaps.SpinOneHeisenberg.conj_twistedHamiltonian,
    CertifiedGaps.SpinOneHeisenberg.partitionFunction_eq_partitionFunctionZ,
    CertifiedGaps.SpinOneHeisenberg.twistedPartitionFunction_eq_gibbsTrace_twistedHamiltonianZ}
  \leanok
  \uses{def:heisenberg}
  The matrix \texttt{exchangeZ} has diagonal \(m_sm_t\), with \(m_s=s-1\), and entry \(1\)
  for each allowed exchange \((m_s,m_t)\mapsto(m_s\pm1,m_t\mp1)\).
  The tensor power of \texttt{szBasis} is unitary.
  For \(L\ne1\), it conjugates the Cartesian ring and its \(\pi\)-twisted ring to the corresponding
  integer \(S^z\) Hamiltonians. Their Gibbs traces agree at every inverse temperature.
\end{lemma}
\begin{proof}
  \leanok
  Check the two-site identities and lift the change of basis to each bond of the ring.
\end{proof}

\begin{lemma}[Energy bounds for twisted rings]
  \label{lem:spin1-energy}
  \lean{CertifiedGaps.SpinOneHeisenberg.exchangeInt_poly,
    CertifiedGaps.SpinOneHeisenberg.abs_qform_triad,
    CertifiedGaps.SpinOneHeisenberg.le_qform_rotTwistedHamiltonian,
    CertifiedGaps.SpinOneHeisenberg.qform_rotTwistedHamiltonian_le}
  \leanok
  \uses{lem:spin1-rotations, lem:spectral-calculus}
  The exchange satisfies \((h-I)(h+I)(h+2I)=0\).
  Two adjacent bonds on three distinct sites have quadratic form bounded in modulus by
  \(3\|x\|^2\).
  If \(R^{\mathsf T}R=I\) and \(R\) preserves the cross product, the ring with closing bond
  \(h(R)\) obeys \(H_L(R)\ge-3L/2\) for \(L\ge3\) and \(H_L(R)\le L\) for \(L\ge2\).
\end{lemma}
\begin{proof}
  \leanok
  A weighted Schur bound proves the three-site estimate. Rotate pairs touching the seam,
  sum the lower bounds over all adjacent pairs, and use the exchange polynomial for the upper
  bound. This supplies the contraction used in \cite[Appendices A and B]{OpenAIHaldane}.
\end{proof}

\begin{lemma}[The two spin-1 sector representations]
  \label{lem:spin1-sectors}
  \lean{CertifiedGaps.SpinOneHeisenberg.exists_sector_approx,
    CertifiedGaps.SpinOneHeisenberg.cTwistedHamiltonian,
    CertifiedGaps.SpinOneHeisenberg.cTwistedHamiltonian_isHermitian,
    CertifiedGaps.SpinOneHeisenberg.exists_ion_approx,
    CertifiedGaps.SpinOneHeisenberg.gibbsTrace_rotTwisted_cRot_transpose}
  \leanok
  \uses{lem:single-site-sectors, lem:spin1-rotations, def:heisenberg-spectrum}
  For \(\beta>0\) and any finite set of lengths \(L\ge2\), the untwisted and \(\pi\)-twisted
  spin-1 traces admit the \(D_2\) approximation.
  Adding the \(2\pi/3\)-twisted trace gives the three-sector approximation.
  The cyclic-twist Hamiltonian is Hermitian. For \(L\ne1\), reversing the cyclic twist leaves
  its Gibbs trace unchanged.
\end{lemma}
\begin{proof}
  \leanok
  The Cartesian \(\pi\) rotations and cyclic permutation of axes realize the required \(A_4\)
  symmetry. A real rotation conjugates the cyclic twist to its inverse.
  These are finite Trotter approximations to the sector traces of
  \cite[Sections 3 and 5.1]{OpenAIHaldane}.
\end{proof}

  consist of the transfer-operator moment inequalities of Definition~\ref{def:moment-bounds} above
  length \(n_0\) at every \(\beta > 0\) together with the symmetric seed purity bounds
  \(\Ps(n_0, \beta_0) \ge 1 - \varepsilon_0\) and \(\Pt(2n_0, \beta_0) \ge 1 - \varepsilon_0\).
\end{definition}

\begin{theorem}[Uniform spectral gap from symmetric seed purities]
  \label{thm:heisenberg-gap-of-inputs}
  \lean{CertifiedGaps.SpinOneHeisenberg.gapGT_of_inputs,
    CertifiedGaps.SpinOneHeisenberg.gapGT_log_twenty}
  \leanok
  \uses{def:thermal-mpo-inputs, thm:gap-of-seeds, lem:heisenberg-trace}
  Let \(n_0 > 0\) be even, \(\beta_0 > 0\), and \(0 < \varepsilon_0 \le 1/100\). Under the seed and
  moment hypotheses of Definition~\ref{def:thermal-mpo-inputs} at \((n_0, \beta_0)\) with defect
  \(\varepsilon_0\), every even \(L \ge n_0\) has a unique ground state and a spectral gap greater
  than \(\log(1/(5\varepsilon_0))/\beta_0\). In particular, at
  \((n_0, \beta_0, \varepsilon_0) = (2304, 784, 1/100)\), every even \(L \ge 2304\) has a spectral
  gap greater than \(\log 20 / 784\).
\end{theorem}
\begin{proof}
  \leanok
  Apply Theorem~\ref{thm:gap-of-seeds}, identifying the matrix-trace partition functions with the
  Boltzmann sums by Lemma~\ref{lem:heisenberg-trace}.
\end{proof}

\section{Scale doubling from length 36 and the Haldane gap}

The proof of Proposition~5.5 of~\cite{OpenAIHaldane} starts from two defect bounds on the rings
of \(36\) and \(72\) sites at \(\beta = 49/4\) (Lemma~\ref{lem:heisenberg-seed36}). Six asymmetric
scale-doubling steps carry them to seeds at \((n_0, \beta_0) = (2304, 784)\)
(Lemma~\ref{lem:heisenberg-chain36}). With the moment inequalities
(Theorem~\ref{thm:heisenberg-moments}), these seeds give the bound~\eqref{eq:haldane-gap}.
The Lean theorem behind Theorem~\ref{thm:haldane-gap} takes another path from the same two
defect bounds: ten steps give a larger gap for every even \(L \ge 36\)
(Theorem~\ref{thm:kernel-gap36-table}), and that gap exceeds \((\log 20)/784\).

\begin{lemma}[Initial defect bounds at \(L = 36\)]
  \label{lem:heisenberg-seed36}
  \lean{CertifiedGaps.SpinOneHeisenberg.seed36_of_table}
  \leanok
  \uses{lem:spin1-certificate36, lem:kernel-spin1-trials, thm:spin1-table36}
  Assume \texttt{TraceTable}. At \(\beta = 49/4\),
  \begin{equation}
    \label{eq:heisenberg-seed36}
    1 - \Ps(36, \beta) \le 0.047916, \qquad 1 - \Pt(72, \beta) \le 0.181521.
  \end{equation}
  These rounded bounds are non-strict in Lean; \cite[Section 5.3]{OpenAIHaldane} obtains
  strict bounds from its unrounded expressions.
\end{lemma}
\begin{proof}
  \leanok
  Insert the kernel-checked trial tensors into Lemma~\ref{lem:spin1-certificate36}.
  Theorem~\ref{thm:spin1-table36} reduces the remaining table hypothesis to
  \texttt{HornerTotals}. The lemmas \texttt{seed36\_spatial} and \texttt{seed36\_thermal}
  in \texttt{Main.lean} are still \texttt{sorry}; this theorem does not use them.
\end{proof}

\begin{lemma}[Six scale-doubling steps from \(L = 36\) to \(L = 2304\)]
  \label{lem:heisenberg-chain36}
  \lean{CertifiedGaps.SpinOneHeisenberg.chain36, CertifiedGaps.SpinOneHeisenberg.chain36_stepsValid,
    CertifiedGaps.SpinOneHeisenberg.seeds2304_of_seeds36,
    CertifiedGaps.purities_ge_chainCertificate, CertifiedGaps.ChainCertificate.StepsValid}
  \leanok
  \uses{lem:chain-purities, def:chain-certificate}
  The initial defect bounds~\eqref{eq:heisenberg-seed36} of Lemma~\ref{lem:heisenberg-seed36} imply
  \(\Ps(2304, 784) \ge 0.99\) and \(\Pt(4608, 784) \ge 0.99\).
\end{lemma}
\begin{proof}
  \leanok
  Six steps of the asymmetric doubling recurrence (Lemma~\ref{lem:chain-purities}), each rounded up
  to a multiple of \(10^{-7}\), reach the defects \((0.0084709, 0.0055149)\) at
  \((36 \cdot 2^6, \tfrac{49}{4} \cdot 2^6) = (2304, 784)\). The rational step
  inequalities~\eqref{eq:asym-step} of Definition~\ref{def:chain-certificate} hold by exact rational
  arithmetic. The paper propagates its exact initial expressions and reaches
  \((0.0084513,0.0054986)\); Lean starts from the rounded bounds in
  Lemma~\ref{lem:heisenberg-seed36}, as opposed to the expressions in
  \cite[Section 5.3]{OpenAIHaldane}.
\end{proof}

\begin{theorem}[Haldane gap of the spin-1 Heisenberg ring for \(L \ge 2304\)]
  \label{thm:haldane-gap}
  \lean{CertifiedGaps.SpinOneHeisenberg.gap_log_twenty_of_table}
  \leanok
  \uses{thm:kernel-gap36-table}
  Assume \texttt{TraceTable}. For every even \(L \ge 2304\), the spin-1 ring has a
  unique ground state, and every excited eigenvalue satisfies
  \begin{equation}
    \label{eq:haldane-gap}
    \Delta_L = E_1(L) - E_0(L) > \frac{\log 20}{784} \approx 0.003821.
  \end{equation}
\end{theorem}
\begin{proof}
  \leanok
  The explicit-hypothesis Lean theorem weakens the sharper bound of
  Theorem~\ref{thm:kernel-gap36-table} by a logarithm comparison.
  This differs from the six-step proof of \cite[Proposition 5.5 and equation (1.4)]{OpenAIHaldane},
  whose seed propagation is recorded in Lemma~\ref{lem:heisenberg-chain36}.
\end{proof}

\begin{remark}
  The theorem above uses a trace-table hypothesis. Theorem~\ref{thm:kernel-gap36-integers}
  states it directly from the printed totals. The older theorem
  \texttt{spin\_one\_heisenberg\_gap} of \texttt{Main.lean} still depends on \texttt{sorry} seeds.
\end{remark}

\section{Uniform gap bound from length 60}

The first bound of Theorem~1.1 of~\cite{OpenAIHaldane} covers shorter rings (\(L \ge 60\)) with a
smaller gap prefactor, starting from a seed with defect \(1/8\) (Lemma~\ref{lem:heisenberg-seed60}),
well above \(1/100\). It follows from the general symmetric spacetime-bootstrap criterion
(Theorem~\ref{thm:gap-of-coefficient}) with \(u_0 = 1/8\) and quadratic prefactor \(C = 79/10\)
(Theorem~\ref{thm:haldane-gap-sixty}).

\begin{lemma}[Initial purity bounds at \(L = 60\)]
  \label{lem:heisenberg-seed60}
  \lean{CertifiedGaps.SpinOneHeisenberg.seed60_of_table}
  \leanok
  \uses{lem:spin1-certificate60, lem:kernel-spin1-trials, thm:spin1-table60, lem:spin1-sectors}
  Assume \texttt{TraceTable60} for the cyclic-twist partition function.
  At \(\beta = 105/4\), \(\Ps(60, \beta) \ge 7/8\) and \(\Pt(120, \beta) \ge 7/8\).
\end{lemma}
\begin{proof}
  \leanok
  The sector and Boltzmann representations hold by Lemma~\ref{lem:spin1-sectors}.
  The kernel-checked trial state supplies the energy input of
  Lemma~\ref{lem:spin1-certificate60}. This proves the non-strict bounds needed from
  \cite[Proposition 5.4]{OpenAIHaldane}; the paper's proof gives strict margins above \(7/8\).
\end{proof}

\begin{theorem}[Haldane gap of the spin-1 Heisenberg ring for \(L \ge 60\)]
  \label{thm:haldane-gap-sixty}
  \lean{CertifiedGaps.SpinOneHeisenberg.gapGT_sixty_of_seeds,
    CertifiedGaps.SpinOneHeisenberg.gap_sixty_of_table}
  \leanok
  \uses{thm:heisenberg-moments, lem:heisenberg-seed60, thm:gap-of-coefficient}
  Assume \texttt{TraceTable60} for the cyclic-twist partition function.
  For every even \(L \ge 60\), the periodic spin-1 Heisenberg ring~\eqref{eq:heisenberg-ham} has a
  unique ground state, and every excited eigenvalue lies more than
  \(\tfrac{4}{105}\log\tfrac{80}{79}\) above the ground energy.
\end{theorem}
\begin{proof}
  \leanok
  Apply Theorem~\ref{thm:gap-of-coefficient} with \(u_0 = 1/8\) and \(C = 79/10\) using
  Theorem~\ref{thm:heisenberg-moments} and Lemma~\ref{lem:heisenberg-seed60}: here
  \(G(1/8) = 913952/117649 \le 79/10\) in~\eqref{eq:doubling-coeff}, \(r = 79/80\) and
  \(C(1 - 3u_0) = 79/16 > 3\), and \(\log(1/r)/\beta_0 = \tfrac{4}{105}\log\tfrac{80}{79}\).
\end{proof}
