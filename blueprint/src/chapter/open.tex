\chapter{Open chains with boundary fields}
\label{chap:open}

The open spin-1 Heisenberg chain with equal boundary fields \(h = 3/5\),
\begin{equation}
  \label{eq:open-heisenberg-ham}
  H_n = \sum_{i=0}^{n-2} \mathbf S_i \cdot \mathbf S_{i+1} - h\,(S_0^z + S_{n-1}^z),
\end{equation}
has, under the finite integer hypotheses below, a unique ground state and a spectral gap
\(\Delta_n > (\log 10)/392 \approx 0.005874\) for every odd \(n \ge 1921\)
(Theorem~\ref{thm:boundary-field-gap}, establishing the theorem of~\cite{OpenAIBoundary}
from its four finite-size seed inputs). Those inputs follow from the periodic trace table,
the fifteen boundary traces, and kernel-checked trial vectors. The final theorem retains only
the sixteen periodic integer totals with \(n \ge 5\) and the six boundary integers at
lengths \(5,6\) as hypotheses (Theorem~\ref{thm:kernel-boundary-integers}); the kernel
checks the other nine boundary integers. The criterion is proved for general seeds and a
rational certificate checked in the kernel (Theorem~\ref{thm:open-criterion},
Definition~\ref{def:open-certificate}); from the same four inputs, a sharper certificate gives
\(\Delta_n > 0.01556\) (Theorem~\ref{thm:boundary-field-gap-sharp}).

\section{Transfer-operator boundary moments}

At every finite Trotter number \(M\), the ring partition functions and the open-chain partition
functions are spectral moments of a single real symmetric transfer matrix \(X = \mathcal F J\),
where \(\mathcal F\) is the single-cell transfer matrix~\eqref{eq:cell-transfer} of
Definition~\ref{def:cell-transfer} and \(J\) reverses imaginary time:
\(Z_{2N} = \Tr X^{2N}\) and \(G_n = \langle v, X^n v\rangle\) for a real boundary vector \(v\)
(Theorem~\ref{thm:open-moment-bounds}). Choosing equal boundary fields at odd chain lengths and
opposite boundary fields at even chain lengths gives both parities this bilinear moment
representation with the same boundary vector \(v\), yielding the closed boundary moment
inequalities~\eqref{eq:open-moment-ineq} (Definition~\ref{def:open-moment-bounds} and
Lemma~\ref{lem:open-moment-bounds-limit}).

\begin{definition}[Transfer-operator boundary moment inequalities]
  \label{def:open-moment-bounds}
  \lean{CertifiedGaps.OpenMomentWitness, CertifiedGaps.OpenMomentBounds,
    CertifiedGaps.leadingRoot, CertifiedGaps.leadingOverlap, CertifiedGaps.boundaryDefect}
  \leanok
  For ring partition functions \(Z\) and open-chain partition functions \(G\), a boundary-moment
  witness \((\mu, c, \rho, s)\) consists of a leading transfer eigenvalue \(\mu \ge 0\), a leading
  boundary overlap \(c \ge 0\), a subleading spectral ratio \(\rho \in [0, 1]\), and a leading
  parity sign \(s = \pm 1\) satisfying the closed moment inequalities induced by a Hermitian
  transfer operator: in particular, \(c\mu^{2N} \le G_{2N}\) and, for \(K \le N\),
  \begin{equation}
    \label{eq:open-moment-ineq}
    \begin{aligned}
      \mu^{2K}(G_{2N} - c\mu^{2N}) & \le \rho^{2N-2K}\mu^{2N}(G_{2K} - c\mu^{2K}), \\
      |G_{2N+1} - sc\mu^{2N+1}|    & \le \mu^{2N+1-2K}(G_{2K} - c\mu^{2K}).
    \end{aligned}
  \end{equation}
  The \emph{boundary defect} is \(e_k = 1 - c\mu^k/G_k\), where \(\mu = \inf_N Z_{2N}^{1/(2N)}\)
  and \(c = \inf_N G_{2N}/\mu^{2N}\).
\end{definition}

\begin{lemma}[Boundary moment inequalities for Hermitian matrices and their limits]
  \label{lem:open-moment-bounds-limit}
  \lean{CertifiedGaps.OpenMomentBounds.of_eigen, CertifiedGaps.OpenMomentBounds.of_hermitian,
    CertifiedGaps.OpenMomentBounds.of_tendsto, CertifiedGaps.OpenMomentWitness.leadingRoot_eq,
    CertifiedGaps.OpenMomentWitness.leadingOverlap_eq}
  \leanok
  \uses{def:open-moment-bounds}
  The moments \(\Tr X^{2N}\) and \(\langle v, X^n v\rangle\) of any Hermitian matrix \(X\) satisfy
  the boundary moment inequalities~\eqref{eq:open-moment-ineq} of
  Definition~\ref{def:open-moment-bounds}, and these inequalities are preserved under pointwise
  limits with \(Z_2, Z_4 > 0\). Every witness satisfies \(\mu = \inf_N Z_{2N}^{1/(2N)}\), and
  \(c = \inf_N G_{2N}/\mu^{2N}\) whenever \(\rho < 1\).
\end{lemma}
\begin{proof}
  \leanok
  Take \(\mu\) to be the largest eigenvalue modulus of \(X\), \(c\) its boundary overlap weight,
  and \(\rho\mu\) the next largest eigenvalue modulus; each inequality
  in~\eqref{eq:open-moment-ineq} holds termwise in the spectral decomposition. For pointwise
  limits, the witnesses lie in a compact box determined by \(Z_2\), \(Z_4\), and \(G_0\), so a
  convergent subsequence yields a limiting witness because each inequality is closed.
\end{proof}

\begin{theorem}[Boundary moments of reflection-positive open chains]
  \label{thm:open-moment-bounds}
  \lean{CertifiedGaps.RPRing.Boundary, CertifiedGaps.RPRing.openHamiltonian,
    CertifiedGaps.RPRing.openTransfer_isHermitian, CertifiedGaps.RPRing.trace_openSlice_pow,
    CertifiedGaps.RPRing.openMomentBounds}
  \leanok
  \uses{lem:open-moment-bounds-limit, def:rpring, def:cell-transfer, lem:chernoff}
  For a reflection-positive ring (Definition~\ref{def:rpring}) with trivial internal reflection
  (\(F = 1\)) and a boundary operator \(b = -\sum_\alpha w_\alpha C_\alpha\) whose coefficients
  \(w_\alpha\) are real for symmetric \(C_\alpha\) and purely imaginary for antisymmetric
  \(C_\alpha\), the ring partition functions and the partition functions of the open chains with
  boundary operators \((b, b)\) at odd length and \((b, b^{\mathsf T})\) at even length satisfy the
  boundary moment inequalities of Definition~\ref{def:open-moment-bounds} at every \(\beta > 0\).
\end{theorem}
\begin{proof}
  \leanok
  Expanding the \(M\)-slice Trotter approximant of the open chain gives the alternating product
  \(c_L^{\mathsf T} \mathcal F^{\mathsf T} \mathcal F \mathcal F^{\mathsf T} \cdots c_R\), where
  \(\mathcal F\) is the single-cell transfer matrix~\eqref{eq:cell-transfer} of
  Definition~\ref{def:cell-transfer}. Reversing the imaginary-time sequences via the involution
  \(J\) satisfies \(\mathcal F^{\mathsf T} = J \mathcal F J\), so \(X = \mathcal F J\) is real
  symmetric and the \(n\)-site alternating product becomes \(J X^n J\) or \(J X^n\). Because the
  coefficient condition on \(b\) implies \(J c_L = \overline{c_L}\), both parities take the
  bilinear form \(\langle v, X^n v\rangle\) while the ring traces equal \(\Tr X^{2N}\). Passing to
  the \(M \to \infty\) limit via Lemma~\ref{lem:open-moment-bounds-limit} and the first-order
  exponential product limit (Lemma~\ref{lem:chernoff}) completes the proof.
\end{proof}

\section{The open-chain spacetime-bootstrap criterion}

Propagating the ring purities and the open-chain boundary defect across doubled inverse
temperatures via~\eqref{eq:open-moment-ineq} yields a uniform lower bound on the spectral gap for
all sufficiently long open chains, odd or even (Theorem~\ref{thm:open-criterion}). The criterion
is stated for general seeds: a seed inverse temperature \(\beta_0\), ring seeds at lengths
\(2N_0\) and \(4N_0\), a boundary seed at length \(2K_0\), and an open-chain seed at length
\(4N_0\). The scales are \(\beta_i = \beta_0 2^i\) and \(N_i = N_0 2^i\); at scale \(i\) the
boundary defect is tracked at length \(2K_i\), with \(K_i = N_i\) for \(i \ge 1\), and the
open-chain thermal purity at length \(4N_i\).

\begin{theorem}[Open-chain spacetime-bootstrap criterion]
  \label{thm:open-criterion}
  \lean{CertifiedGaps.OpenMomentWitness.move_one, CertifiedGaps.OpenMomentWitness.move_two,
    CertifiedGaps.OpenMomentWitness.oddChain_purity,
    CertifiedGaps.OpenMomentWitness.evenChain_purity, CertifiedGaps.bulk_pow_le,
    CertifiedGaps.open_step, CertifiedGaps.boundaryHalf, CertifiedGaps.OpenStep,
    CertifiedGaps.gapGT_of_openChain}
  \leanok
  \uses{def:open-moment-bounds, lem:doubling-coefficient, lem:purity-squaring, lem:purity-to-gap}
  Assume the transfer-operator moment inequalities of Definition~\ref{def:moment-bounds} for the
  ring and the boundary moment inequalities of Definition~\ref{def:open-moment-bounds} at every
  \(\beta > 0\). Suppose that
  \begin{enumerate}
    \item the ring satisfies \(\Ps(2N_0, \beta_0), \Pt(4N_0, \beta_0) \ge 1 - u_0\) with
      \(u_0 \le 1/10\), \(G(u_0) \le C \le 6\), and \(Cu_0 \le 1\), so that the ring defects are
      \(\varepsilon_i = (Cu_0)^{2^i}/C\) (Lemma~\ref{lem:doubling-coefficient});
    \item the boundary defect satisfies \(e_{2K_0}(\beta_0) \le E_0\), and the opposite-field
      open-chain thermal purity satisfies \(\Pt(4N_0, \beta_0) \ge 1 - \eta_0\);
    \item \(m \ge 1\) base steps \((E_i, \eta_i) \mapsto (E_{i+1}, \eta_{i+1})\) with suppressions
      \(q_i\) satisfy \(d(\varepsilon_{i+1})^{a_i/g_i} \le q_i^{b_i/g_i}\), where
      \(a_i = 4N_i - 2K_i\), \(b_i = 4N_i\), and \(g_i = \gcd(a_i, b_i)\), together with the
      overlap update \(F(q_i, E_i, 1 - \eta_i) \le E_{i+1}\) and the squaring bound
      \(f\bigl(1 - (1 - \eta_i)(1 - \varepsilon_i)(1 - \varepsilon_{i+1})(1 - E_{i+1})\bigr)
      \le \eta_{i+1}\);
    \item the hand-over defect \(x\) satisfies \(E_m, \eta_m, \varepsilon_m \le x \le 1/100\).
  \end{enumerate}
  Then every open chain of \(n \ge 4N_0 2^m\) sites, odd or even, has a unique ground state and a
  spectral gap greater than \(\log(1/(10x))/(\beta_0 2^m)\).
\end{theorem}
\begin{proof}
  \leanok
  By Lemma~\ref{lem:doubling-coefficient}, the ring purities hold at every scale, and they control
  both the subleading spectral ratio, \(\rho_i^{2N_i} \le d(\varepsilon_i)\), and the bulk
  normalization ratio \(\mu_{i+1}/\mu_i^2\) in Definition~\ref{def:open-moment-bounds}. Three
  recurrence steps advance the boundary defect and the open-chain thermal purity from scale \(i\)
  to scale \(i + 1\) using~\eqref{eq:open-moment-ineq} and Lemma~\ref{lem:purity-squaring}: the
  overlap recurrence, length doubling, and thermal squaring. The base steps reach scale \(m\), after
  which all defects at scale \(m + j\) lie below \(x_j = (10x)^{2^j}/10\) with
  \(x_{j+1} = 10x_j^2\). At odd lengths the leading parity sign in~\eqref{eq:open-moment-ineq} is
  \(s = +1\) because the partition function is positive; at even lengths there is no sign, and the
  decay inequality alone bounds the trace. Either way \(\Pt(n, \beta_{m+j}) \ge 1 - 4x_j\)
  on the window \(4N_{m+j} \le n \le 8N_{m+j}\) and hence the uniform gap bound
  via~\eqref{eq:purity-to-gap} in Lemma~\ref{lem:purity-to-gap}.
\end{proof}

\begin{lemma}[The boundary seed from finite traces]
  \label{lem:boundary-seed-traces}
  \lean{CertifiedGaps.boundaryDefect_le_of_traces}
  \leanok
  \uses{def:open-moment-bounds}
  Assume the boundary moment inequalities at \(\beta\) and \(\Ps(2N_0, \beta) \ge 1 - \varepsilon\)
  with \(\varepsilon \le 1/10\). If \(N_0 r < K\) and
  \(G_{2K - 2N_0 r}\,\bigl(d(\varepsilon) Z_{2N_0}\bigr)^r \le E\, G_{2K}\), then
  \(e_{2K} \le E\).
\end{lemma}
\begin{proof}
  \leanok
  The decay inequality~\eqref{eq:open-moment-ineq} at \(2K' = 2K - 2N_0 r\) gives
  \(G_{2K} - c\mu^{2K} \le (\rho\mu)^{2N_0 r} G_{2K'}\), and the ring purity gives
  \(\rho^{2N_0} \le d(\varepsilon)\) and \(\mu^{2N_0} \le Z_{2N_0}\). So the boundary seed, which
  involves the leading eigenvalue and overlap, follows from three finite partition functions.
\end{proof}

    \mu\ge0.999,\qquad \rho\le0.874,\qquad cP_*(\mu)^2\le0.003745.
  \]
  Together with \(Z_{6,1}<0.027\), this implies \texttt{ThermalCore}.
  The bounds are non-strict in Lean, unlike the strict inequalities of
  \cite[Proposition 3.4 and Proposition A.1]{OpenAIBoundary}.
\end{theorem}
\begin{proof}
  \leanok
  Apply the projected polynomial bound before taking the Trotter limit.
  Compactness preserves the closed witness inequalities.
  Positivity of the odd open traces removes the leading sign (or the overlap is zero);
  the paper fixes the sign using ring traces.
\end{proof}

\section{Initializing the two open-chain seeds}

The leading witness controls upper trace bounds; variational states give lower ground weights.
Two boundary updates then reach length \(142\) and inverse temperature \(49\).

\begin{definition}[Thermal data and four trial bounds]

  \texttt{TrialData} requires an opposite-field eigenvalue below
  \(-am+2021/5000\) at each \(m=34,78,142,240\).
\end{definition}

\begin{lemma}[The opposite-field variational bridge]
  \label{lem:boundary-trial-bridge}
  \lean{CertifiedGaps.SpinOneHeisenberg.conj_endFieldHamiltonian,

\end{lemma}
\begin{proof}
  \leanok

\end{proof}

\section{The open spin-1 Heisenberg chain with boundary fields}

Specializing Corollary~\ref{cor:open-seeds-note} to the Cartesian spin-1 generators of
Definition~\ref{def:spin-one-generators} with boundary fields \(h = 3/5\) establishes the open-chain
Haldane gap bound (Theorem~\ref{thm:boundary-field-gap}) from four finite-size seed inputs
(Lemma~\ref{lem:open-seeds}).

\begin{lemma}[Finite-size seed bounds for the boundary-field chain]
  \label{lem:open-seeds}
  \lean{CertifiedGaps.SpinOneHeisenberg.BoundaryFieldInputs,
    CertifiedGaps.SpinOneHeisenberg.BoundaryInit.boundaryDefect_le_of_table,
    CertifiedGaps.SpinOneHeisenberg.BoundaryInit.oppositeFieldThermalPurity_ge_of_table}
  \leanok
  \uses{lem:boundary-reseed120, lem:boundary-initialized, def:boundary-init-data}
  Assume \texttt{ThermalCore}, \texttt{TrialData}, \texttt{TraceTable}, and the two periodic
  trial hypotheses of Lemma~\ref{lem:boundary-reseed120}.
  At \(\beta = 49\), the periodic spin-1 Heisenberg ring (Definition~\ref{def:heisenberg-spectrum})
  satisfies \(\Ps(120, 49) \ge 0.948\) and \(\Pt(240, 49) \ge 0.948\), and the opposite-field open
  chain with \(h = 3/5\) satisfies \(e_{142} \le 0.14\) and \(\Pt(240, 49) \ge 0.79\).
\end{lemma}
\begin{proof}
  \leanok
  Reseed the ring by Lemma~\ref{lem:boundary-reseed120}, supply the upper spectral window from
  the table, and apply Lemma~\ref{lem:boundary-initialized}.
  These four inequalities form \(\texttt{BoundaryFieldInputs}(3/5)\).
  Theorem~\ref{thm:kernel-boundary-inputs} supplies them from the remaining printed-integer
  hypotheses. The four lemmas \texttt{seed\_*} in \texttt{OpenChain.lean} are still
  \texttt{sorry}; this route does not use them.
\end{proof}

\begin{theorem}[Gap for equal boundary fields]
  \label{thm:boundary-field-gap}
  \lean{CertifiedGaps.SpinOneHeisenberg.endFieldHamiltonian,
    CertifiedGaps.SpinOneHeisenberg.openHamiltonian_odd,
    CertifiedGaps.SpinOneHeisenberg.openHamiltonian_even,
    CertifiedGaps.SpinOneHeisenberg.gapGT_of_boundaryFieldInputs,
    CertifiedGaps.SpinOneHeisenberg.gapGT_of_boundaryFieldInputs_certificate}
  \leanok
  \uses{cor:open-seeds-note, lem:open-seeds}
  Assume the four inequalities in Lemma~\ref{lem:open-seeds},
  bundled as \(\texttt{BoundaryFieldInputs}(3/5)\).
  For every odd \(n \ge 1921\), the open spin-1 Heisenberg chain~\eqref{eq:open-heisenberg-ham} with
  equal boundary fields \(h = 3/5\) has a unique ground state, and every excited eigenvalue lies
  more than \((\log 10)/392\) above the ground energy.
\end{theorem}
\begin{proof}
  \leanok
  In the Cartesian basis (Definition~\ref{def:spin-one-generators}), taking the imaginary
  coefficient \(w_z = -ih\) on the antisymmetric generator \(G_z = i S^z\) produces the boundary
  operator \(b = -h S^z\), whose transpose is \(b^{\mathsf T} = +h S^z\); the open chain of
  Theorem~\ref{thm:open-moment-bounds} therefore has equal boundary fields
  \eqref{eq:open-heisenberg-ham} at odd length and opposite boundary fields at even length.
  Combining Theorem~\ref{thm:rp-moment-bounds}, Theorem~\ref{thm:open-moment-bounds}, and
  Corollary~\ref{cor:open-seeds-note} with the inputs of Lemma~\ref{lem:open-seeds} gives the
  bound.
\end{proof}

\begin{theorem}[A sharper gap from the same seeds]
  \label{thm:boundary-field-gap-sharp}
  \lean{CertifiedGaps.sharpBoundaryFieldCertificate,
    CertifiedGaps.sharpBoundaryFieldCertificate_valid,
    CertifiedGaps.SpinOneHeisenberg.gapGT_of_boundaryFieldInputs_sharp}
  \leanok
  \uses{def:open-certificate, lem:open-seeds, thm:boundary-field-gap}
  From the inputs of Lemma~\ref{lem:open-seeds}, every odd \(n \ge 1921\) has a unique ground state
  and a spectral gap greater than \(\log(10^6/2241)/392 \approx 0.01556\).
\end{theorem}
\begin{proof}
  \leanok
  The certificate of Corollary~\ref{cor:open-seeds-note} is conservative. Taking
  \(C = 5.583 \ge G(0.052)\) in place of \(6\), four-digit bounds along the same three base steps,
  and the hand-over at the defect reached at scale \(3\), \(x = 2.241 \cdot 10^{-4}\), in place of
  \(1/100\), Definition~\ref{def:open-certificate} gives the bound \(\log(1/(10x))/392\).
  This sharpens the constant \((\log 10)/392\) of \cite[Theorem 1.1]{OpenAIBoundary}, which
  hands over at \(x = 1/100\) \cite[Sections 5.3 and 6]{OpenAIBoundary}.
\end{proof}


\begin{theorem}[The opposite-field chain at even lengths]
  \label{thm:opposite-field-gap}
  \lean{CertifiedGaps.SpinOneHeisenberg.oppositeFieldSpectrum,
    CertifiedGaps.SpinOneHeisenberg.openSpectrum_even,
    CertifiedGaps.SpinOneHeisenberg.openGapGT_of_boundaryFieldInputs_certificate,
    CertifiedGaps.SpinOneHeisenberg.oppositeFieldGapGT_of_boundaryFieldInputs_certificate,
    CertifiedGaps.SpinOneHeisenberg.oppositeFieldGapGT_of_boundaryFieldInputs,
    CertifiedGaps.SpinOneHeisenberg.oppositeFieldGapGT_of_boundaryFieldInputs_sharp}
  \leanok
  \uses{thm:boundary-field-gap, thm:boundary-field-gap-sharp}
  From the inputs of Lemma~\ref{lem:open-seeds}, for every even \(n \ge 1920\) the open spin-1
  Heisenberg chain with opposite boundary fields,
  \(\sum_{i=0}^{n-2} \mathbf S_i \cdot \mathbf S_{i+1} - h\,(S_0^z - S_{n-1}^z)\) with \(h = 3/5\),
  has a unique ground state and a spectral gap greater than \((\log 10)/392\), and greater than
  \(\log(10^6/2241)/392\) with the certificate of Theorem~\ref{thm:boundary-field-gap-sharp}.
\end{theorem}
\begin{proof}
  \leanok
  At even length the open chain of Theorem~\ref{thm:boundary-field-gap} is the opposite-field chain,
  and Theorem~\ref{thm:open-criterion} covers even lengths.
  This adds an even-length conclusion to the odd equal-field theorem of
  \cite[Theorem 1.1]{OpenAIBoundary}.
\end{proof}
